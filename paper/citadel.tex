	\frame{\titlepage}
	
	\begin{frame}
		\frametitle{What is Citadel}
		\begin{itemize}
			\item \textbf{Sovereign Search Engine:} A purely private, closed-loop search environment built exclusively on user-provided data.
			\item \textbf{High-Fidelity Intelligence:} Advanced analytical platform ensuring flawless retrieval accuracy across massive file scales.
			\item \textbf{End-to-End Workflow:} Ingest arbitrary documents to instantly query, extract insights, and auto-generate comprehensive reports.
			\item \textbf{High-Throughput Ingestion:} Seamlessly processes hundreds of multi-page, complex files in minutes.
			\item \textbf{Multi-Modal Structural Extraction:} Creates structure from completely unstructured visual, text, and layout media.
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Defeating the Flaws of Frontier Market Solutions}
		\begin{itemize}
			\item \textbf{Global Cross-Document Synthesis:} Succeeds at macroscopic aggregation, cross-file comparisons, and trend filtering where traditional semantic search completely breaks.
			\item \textbf{Tabular Precision:} Even the most modern document intelligence systems flatten tables into linear strings, breaking cell connections. Citadel not only preserves the two-dimensional relational geometry of tabular data but also maintains the root of origin for reference.
			\item \textbf{Context Window Optimization:} Eliminates prompt bloat, intelligently isolates data rather than exploding token limits.
			\item \textbf{No Lost in the Middle Failures:} Circumvents the critical retrieval degradation that plagues standard multi-billion dollar cloud architectures.
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Why it Works}
		\textbf{Moving Beyond Legacy AI Framework Paradigms:}
		\vspace{0.3cm}
		\begin{itemize}
			\item \textbf{Transcending Traditional RAG:} Not a naive vector search, immune to typical RAG lookup bottlenecks and token limitations.
			\item \textbf{Surpassing Rigid Knowledge Graphs:} Eliminates complex and rigid schema-mapping latencies and entity-relationship or triplet extraction failures.
			\item \textbf{Granular Level Traceability:} Every generation, data point, and report metric features a direct, verifiable mathematical path back to its exact source coordinate.
			\item \textbf{Ingestion Architecture:} No fixed-size sliding windows or token chunking, completely eliminating context fracturing and boundary loss.
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Degree of Freedom}
		\begin{itemize}
			\item \textbf{Zero Hard-Coded Scaffolding:} No hidden, internal prompt engineering forcing specific outputs or introducing algorithmic bias.
			\item \textbf{Complete Execution Sovereignty:} The user maintains absolute, uncapped control over how their data is manipulated, synthesized, structured, and operationalized.
		\end{itemize}
	\end{frame}
\end{document}
